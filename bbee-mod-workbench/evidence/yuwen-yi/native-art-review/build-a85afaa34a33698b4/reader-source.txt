def metric(actual,expected):
    require(actual.shape==expected.shape,'pixel shapes differ')
    delta=actual.astype(np.int16)-expected.astype(np.int16);absolute=np.abs(delta)
    return {'max_rgba':absolute.max(axis=(0,1)).tolist(),'mae_rgba':np.round(absolute.mean(axis=(0,1)),6).tolist(),
            'rmse_rgba':np.round(np.sqrt((delta.astype(float)**2).mean(axis=(0,1))),6).tolist(),
            'changed_pixels':int(np.any(delta,axis=2).sum()),'pixels':int(actual.shape[0]*actual.shape[1]),
            'actual_alpha_nonzero':int((actual[:,:,3]>0).sum()),'expected_alpha_nonzero':int((expected[:,:,3]>0).sum())}
def straight(pixels):

    return Image.fromarray(np.clip(np.rint(a),0,255).astype(np.uint8),'RGBA')

def expected_sequence(plan,total):
    if any('exact_ticks' in x for x in plan):dur=[x['exact_ticks'] for x in plan]
    else:
        ws=[Fraction(x['weight']) for x in plan];budget=total-len(ws);shares=[budget*w/sum(ws) for w in ws]
        dur=[1+s.numerator//s.denominator for s in shares]
        order=sorted(range(len(ws)),key=lambda i:(-(shares[i]%1),i))
        for i in order[:total-sum(dur)]:dur[i]+=1
    require(sum(dur)==total and all(type(n)==int and n>0 for n in dur),'invalid exposure plan')
    result=[]
    for item,n in zip(plan,dur):
        if result and result[-1][0]==item['key']:result[-1][1]+=n
        else:result.append([item['key'],n])
    return result
